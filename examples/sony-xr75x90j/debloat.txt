# system image is never touched, so tools/restore.sh reverses every line below.

#   com.sony.dtv.voicesearch, com.sony.dtv.searchapp             voice assistant  (user uses it)
#   com.sony.dtv.tvx.overlay.sortui, com.sony.dtv.settings.overlay, overlays, com.android.* system plumbing

com.sony.dtv.da.service
com.qterics.da.product
# Interactive TV util: logs app/voice/action usage and ships it out.
com.sony.dtv.interactivetvutil
com.sony.dtv.interactivetvutil.output
# IDD agent: "Improved Device Diagnostics", Sony's crash/usage reporting.
com.sonyericsson.idd.agent
# Sony log level plumbing + bug report service.
com.sony.dtv.sonyloglevelsettingsys
com.sony.dtv.sonybugreportsys
# Last access: per-app usage tracking feeding Sony's recommendation/telemetry backend.
com.sony.dtv.lastaccess
# Kids profile adapter: log upload + account service.
com.sony.dtv.kidsprofileadapter
#
# HELD BACK (see notes at the end of this file): the MediaTek platform agents are NOT listed here
# on purpose. They are PERSISTENT system services, none of them holds android.permission.INTERNET,
# and com.mediatek.agent declares TVRemoteServiceReceiver (com.sony.dtv.tvapi holds its
# BIND_TVREMOTE_SERVICE_AGENT permission). Disabling them risks the remote control, which this
# project must never break, and gains almost nothing: without INTERNET they cannot phone home.
